        function wrapLabel(str, maxWidth) {
            if (typeof str !== 'string') return str; // Handle non-string inputs like arrays from previous wrapping
            if (str.length <= maxWidth) {
                return str;
            }
            const words = str.split(' ');
            let lines = [];
            let currentLine = '';
            for (const word of words) {
                if ((currentLine + word).length > maxWidth && currentLine.length > 0) {
                    lines.push(currentLine.trim());
                    currentLine = word + ' ';
                } else {
                    currentLine += word + ' ';
                }
            }
            lines.push(currentLine.trim());
            return lines;
        }

        const mitigationModal = document.getElementById('mitigationModal');
        const mitigationModalTitle = document.getElementById('mitigationModalTitle');
        const mitigationModalBody = document.getElementById('mitigationModalBody');
        const mitigationModalLoading = document.getElementById('mitigationModalLoading');
        
        function openModal(modalId, titleContent = "Insights") {
            const modal = document.getElementById(modalId);
            if (modalId === 'mitigationModal' && mitigationModalTitle) {
                mitigationModalTitle.textContent = titleContent;
            }
            if(modal) modal.style.display = "block";
        }

        function closeModal(modalId) {
            const modal = document.getElementById(modalId);
            if(modal) modal.style.display = "none";
            if (modalId === 'mitigationModal' && mitigationModalBody) {
                 mitigationModalBody.innerHTML = ''; // Clear previous content
            }
        }
        
        // Close modal if user clicks outside of it
        window.onclick = function(event) {
            if (event.target == mitigationModal) {
                closeModal('mitigationModal');
            }
        }

        async function getMitigationInsights(epochName) {
            openModal('mitigationModal', `Mitigation Insights for ${epochName} Epoch`);
            mitigationModalBody.innerHTML = ''; // Clear previous content
            mitigationModalLoading.classList.remove('hidden');

            const prompt = `
                You are an expert strategist focusing on mitigating Symbolic Collapse Pressure (SCP).
                The SCP framework (C = (I*S*P)/(T*E)) indicates that the "${epochName}" epoch is experiencing high symbolic collapse pressure (C > 1.0).
                This means factors like Information Load (I), Symbolic Abstraction (S), and/or Polarization (P) are high,
                while Transmission Fidelity (T) and/or Epistemic Coherence (E) are low.

                Please suggest 3-4 actionable and creative strategies or policy ideas to help mitigate the high symbolic collapse pressure specifically for the "${epochName}" epoch.
                For each strategy, briefly explain how it would help improve T or E, or reduce I, S, or P.
                Focus on practical and impactful ideas.
            `;
            const resultText = await callGeminiAPI(prompt);
            mitigationModalBody.innerHTML = resultText.replace(/\n/g, '<br>'); // Simple formatting
            mitigationModalLoading.classList.add('hidden');
        }

        function addMitigationButtons() {
            const container = document.getElementById('mitigationInsightsContainer');
            if (!container) return;

            epochRawLabels.forEach((label, index) => {
                const epochName = Array.isArray(label) ? label[0] : label; // Get primary name
                const cValue = epochDataValues[index];

                if (cValue >= 1.0) { // Only for epochs above threshold
                    const buttonDiv = document.createElement('div');
                    buttonDiv.classList.add('text-center', 'mt-4');
                    
                    const button = document.createElement('button');
                    button.classList.add('gemini-button', 'mx-auto');
                    button.innerHTML = `✨ Get Mitigation Insights for ${epochName} Epoch <span id="mitigationLoading-${index}" class="loading-spinner hidden ml-2"></span>`;
                    button.onclick = () => getMitigationInsights(epochName);
                    
                    buttonDiv.appendChild(button);
                    container.appendChild(buttonDiv);
                }
            });
        }

                }
                 new Chart(ctxSimulations, epochalChartConfig);
            } else {
                console.error("Canvas element for epochal simulations chart not found.");
            }
            addMitigationButtons();
